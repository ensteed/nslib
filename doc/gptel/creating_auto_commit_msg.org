*** Is there any way emacs could not block while doing this?

Yes—use an async process and a sentinel/filter, then insert when it finishes. The key is: start =claude= with =make-process= (or =start-process=), capture output in a temp buffer, and on exit insert into the original buffer.

Here’s a drop-in async version that inserts at point without blocking:

#+begin_src elisp
(defvar my/claude--commit-process nil)

(defun my/claude-insert-commit-message-async ()
  "Asynchronously generate a commit message from staged diff and insert at point."
  (interactive)
  (unless (magit-git-repo-p default-directory)
    (user-error "Not in a git repository"))

  (let ((diff (magit-git-string "diff" "--staged")))
    (when (or (null diff) (string-empty-p diff))
      (user-error "No staged changes"))

    (when (process-live-p my/claude--commit-process)
      (user-error "Claude generation already running"))

    (let* ((origin-buf (current-buffer))
           (origin-pt  (point))
           (origin-dir default-directory)
           (out-buf (generate-new-buffer " *claude-commit-output*"))
           ;; Adjust this prompt to your taste:
           (prompt "Generate a concise git commit subject line for these staged changes:\n\n")
           ;; Some CLIs want the prompt as an arg. If yours differs, change :command below.
           (proc
            (make-process
             :name "claude-commit"
             :buffer out-buf
             :noquery t
             :command (list "claude" (concat prompt diff))
             :sentinel
             (lambda (p _event)
               (when (memq (process-status p) '(exit signal))
                 (let ((exit (process-exit-status p)))
                   (unwind-protect
                       (if (/= exit 0)
                           (message "claude failed (%s): %s" exit
                                    (with-current-buffer (process-buffer p)
                                      (string-trim (buffer-string))))
                         (let ((msg (with-current-buffer (process-buffer p)
                                      (string-trim (buffer-string)))))
                           (when (buffer-live-p origin-buf)
                             (with-current-buffer origin-buf
                               (let ((inhibit-read-only t))
                                 (goto-char origin-pt)
                                 (insert msg)))
                             (message "Inserted Claude commit message."))))
                     (when (buffer-live-p (process-buffer p))
                       (kill-buffer (process-buffer p))))))))))
      (setq my/claude--commit-process proc)
      (with-current-buffer out-buf
        (setq default-directory origin-dir))
      (message "Generating commit message asynchronously..."))))
#+end_src
